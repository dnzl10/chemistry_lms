/* ════════════════════════════════════════════
   FILTERING & SORTING
════════════════════════════════════════════ */
function filterSubject(sid, btn) {
    activeSubject = String(sid);
    document.querySelectorAll('.subject-tab').forEach(b => b.classList.remove('active'));
    btn.classList.add('active');

    // show/hide topic pills
    document.querySelectorAll('#topicFilterRow .filter-pill').forEach(p => {
        const ps = p.dataset.subject;
        if (!ps || activeSubject === 'all' || ps === activeSubject) {
            p.style.display = '';
        } else {
            p.style.display = 'none';
        }
    });
    activeTopic = 'all';
    document.querySelector('#topicFilterRow [data-topic="all"]').classList.add('active');
    document.querySelectorAll('#topicFilterRow [data-topic]:not([data-topic="all"])').forEach(p => p.classList.remove('active'));
    applyFilters();
}

function filterTopic(tid, btn) {
    activeTopic = String(tid);
    document.querySelectorAll('#topicFilterRow .filter-pill').forEach(p => p.classList.remove('active'));
    btn.classList.add('active');
    applyFilters();
}

function togglePill(type, btn) {
    if (type === 'pin') { filterPin = !filterPin; btn.classList.toggle('active', filterPin); }
    applyFilters();
}

function applyFilters() {
    let list = [...ALL_FORUMS];
    const search = document.getElementById('searchInput').value.trim().toLowerCase();
    const sort   = document.getElementById('sortSelect').value;

    if (activeSubject !== 'all') list = list.filter(f => String(f.subject_id) === activeSubject);
    if (activeTopic   !== 'all') list = list.filter(f => String(f.topic_id)   === activeTopic);
    if (filterPin)                list = list.filter(f => f.is_pinned);
    if (search)                   list = list.filter(f => f.title.toLowerCase().includes(search));

    if (sort === 'newest')       list.sort((a,b) => new Date(b.created_at) - new Date(a.created_at));
    else if (sort === 'oldest')  list.sort((a,b) => new Date(a.created_at) - new Date(b.created_at));
    else if (sort === 'most_replies') list.sort((a,b) => (b.reply_count||0) - (a.reply_count||0));
    else if (sort === 'pinned')  list.sort((a,b) => b.is_pinned - a.is_pinned);

    renderForumList(list);
}

/* ════════════════════════════════════════════
   OPEN THREAD (fetch detail)
════════════════════════════════════════════ */
async function openThread(forumId) {
    try {
        const res  = await fetch(`/teacher/api/forum/${forumId}`);
        const data = await res.json();
        if (!res.ok) throw new Error(data.error || 'Failed to load');

        currentForum = data.forum;

        // Switch views
        document.getElementById('forumListView').classList.remove('active');
        document.getElementById('threadView').classList.add('active');
        document.getElementById('mainScroll').scrollTo(0, 0);

        renderThread(data);
    } catch (e) {
        toast(e.message, 'error');
    }
}

function renderThread(data) {
    const f       = data.forum;
    const replies = data.replies || [];
    const done    = data.done_students || [];
    const students = data.enrolled_students || [];

    // Header fields
    document.getElementById('threadTitle').textContent = f.title;
    document.getElementById('threadDate').textContent  = formatDate(f.created_at);
    document.getElementById('threadReplyCount').textContent = f.reply_count || 0;
    document.getElementById('threadDoneCount').textContent  = done.length;
    document.getElementById('threadContent').textContent    = f.forum_content || '';
    document.getElementById('replyCountBadge').textContent  = f.reply_count || 0;

    // Pin badge/button
    const pinBadge = document.getElementById('threadPinBadge');
    const pinLabel = document.getElementById('btnPinLabel');
    pinBadge.classList.toggle('hidden', !f.is_pinned);
    pinLabel.textContent = f.is_pinned ? 'Unpin' : 'Pin';

    // Topic & Subject badges
    const subject = ALL_SUBJECTS.find(s => s.subject_id === f.subject_id) || {};
    const topic   = ALL_TOPICS.find(t => t.topic_id === f.topic_id) || null;
    document.getElementById('threadSubjectBadge').textContent = subject.subject_name || '';
    document.getElementById('threadTopicBadge').textContent   = topic ? `Q${topic.quarter_number} · ${topic.topic_name}` : '';
    document.getElementById('threadTopicBadge').style.display = topic ? '' : 'none';

    // Author
    document.getElementById('threadAuthor').innerHTML = `
        <img src="/profile_uploads/${f.teacher_pic || 'default_avatar.png'}"
             class="avatar w-8 h-8" onerror="this.src='/profile_uploads/default_avatar.png'">
        <div>
            <p class="text-white font-bold text-sm">${escHtml(f.teacher_name || 'Teacher')}</p>
            <p class="text-white/60 text-xs">Posted ${timeAgo(f.created_at)}</p>
        </div>`;

    // Completion tracker
    const enrolled = students.length;
    const doneIds  = new Set(done.map(d => d.student_id));
    const pct      = enrolled ? Math.round((done.length / enrolled) * 100) : 0;
    document.getElementById('completionFill').style.width  = pct + '%';
    document.getElementById('completionLabel').textContent = pct + '%';

    const compList = document.getElementById('completionList');
    if (!enrolled) {
        compList.innerHTML = '<p class="text-xs text-slate-400 py-3 text-center">No students enrolled.</p>';
    } else {
        compList.innerHTML = students.map(s => {
            const isDone = doneIds.has(s.student_id);
            return `<div class="student-done-row">
                <img src="/profile_uploads/${s.profile_pic || 'default_avatar.png'}"
                     class="avatar w-7 h-7"
                     onerror="this.src='/profile_uploads/default_avatar.png'">
                <span class="text-xs font-semibold text-slate-700 flex-1">${escHtml(s.first_name)} ${escHtml(s.last_name)}</span>
                ${isDone
                    ? `<span class="badge badge-emerald"><i class="fas fa-check mr-1"></i> Done <span class="ml-1 opacity-60">${timeAgo(done.find(d=>d.student_id===s.student_id).done_at)}</span></span>`
                    : `<span class="badge badge-slate">Not yet</span>`}
            </div>`;
        }).join('');
    }

    // Replies
    renderReplies(replies);
}

/* ════════════════════════════════════════════
   RENDER REPLIES (nested)
════════════════════════════════════════════ */
function renderReplies(replies) {
    const container = document.getElementById('repliesContainer');
    if (!replies.length) {
        container.innerHTML = `<div class="empty-state" style="padding:24px">
            <i class="fas fa-comment-slash text-slate-300 text-2xl"></i>
            <p class="text-xs text-slate-400">No replies yet. Start the discussion!</p>
        </div>`;
        return;
    }

    // Build tree
    const roots   = replies.filter(r => !r.parent_reply_id);
    const children = {};
    replies.filter(r => r.parent_reply_id).forEach(r => {
        (children[r.parent_reply_id] = children[r.parent_reply_id] || []).push(r);
    });

    function renderReply(r, depth = 0) {
        const isTeacherReply = r.user_id === TEACHER_ID;
        const hasFile = r.file_path && r.file_name;
        const hasLink = r.external_link;
        const cls = `reply-card fade-in ${depth > 0 ? 'nested' : ''} ${isTeacherReply ? 'teacher-reply' : ''}`;

        let attachHtml = '';
        if (hasFile) {
            const icon = fileIcon(r.file_type);
            attachHtml += `<a href="/uploads/forum/${r.file_path}" target="_blank" class="file-preview-box inline-flex mt-2 hover:border-indigo-300 transition">
                <i class="${icon} text-indigo-400 text-lg"></i>
                <div class="flex-1 min-w-0">
                    <p class="font-bold text-slate-700 truncate text-xs">${escHtml(r.file_name)}</p>
                    <p class="text-[10px] text-slate-400">${r.file_type || 'File'}</p>
                </div>
                <i class="fas fa-download text-slate-400 text-xs"></i>
            </a>`;
        }
        if (hasLink) {
            attachHtml += `<a href="${escHtml(r.external_link)}" target="_blank" class="mt-2 flex items-center gap-2 text-xs text-indigo-600 font-semibold hover:underline">
                <i class="fas fa-external-link-alt"></i> ${escHtml(r.external_link.length > 60 ? r.external_link.slice(0,60)+'…' : r.external_link)}
            </a>`;
        }

        const childReplies = (children[r.reply_id] || []);
        const childrenHtml = childReplies.map(c => renderReply(c, depth + 1)).join('');

        return `<div>
            <div class="${cls}" id="reply-${r.reply_id}">
                <div class="flex items-start gap-2">
                    <img src="/profile_uploads/${r.profile_pic || 'default_avatar.png'}"
                         class="avatar w-7 h-7" onerror="this.src='/profile_uploads/default_avatar.png'">
                    <div class="flex-1 min-w-0">
                        <div class="flex items-center gap-2 flex-wrap mb-1">
                            <span class="text-xs font-black text-slate-800">${escHtml(r.author_name || 'Unknown')}</span>
                            ${isTeacherReply ? '<span class="badge badge-indigo">Instructor</span>' : '<span class="badge badge-slate">Student</span>'}
                            <span class="text-[10px] text-slate-400">${timeAgo(r.created_at)}</span>
                        </div>
                        <p class="text-xs text-slate-700 leading-relaxed whitespace-pre-line">${escHtml(r.reply_content)}</p>
                        ${attachHtml}
                    </div>
                    <div class="flex gap-1 flex-shrink-0">
                        <button class="btn btn-outline btn-xs" onclick="openReplyModal(${r.reply_id}, '${escHtml(r.author_name||'').replace(/'/g,"\\'")}')">
                            <i class="fas fa-reply"></i>
                        </button>
                        <button class="btn btn-danger btn-xs" onclick="openDeleteReplyModal(${r.reply_id})">
                            <i class="fas fa-trash"></i>
                        </button>
                    </div>
                </div>
            </div>
            ${childrenHtml ? `<div class="mt-2 flex flex-col gap-2">${childrenHtml}</div>` : ''}
        </div>`;
    }

    container.innerHTML = `<div class="flex flex-col gap-3">${roots.map(r => renderReply(r)).join('')}</div>`;
}

function fileIcon(type) {
    if (!type) return 'fas fa-file';
    if (type.includes('pdf')) return 'fas fa-file-pdf';
    if (type.includes('word') || type.includes('doc')) return 'fas fa-file-word';
    if (type.includes('image') || type.includes('png') || type.includes('jpg')) return 'fas fa-file-image';
    if (type.includes('video')) return 'fas fa-file-video';
    if (type.includes('audio')) return 'fas fa-file-audio';
    if (type.includes('zip') || type.includes('rar')) return 'fas fa-file-archive';
    return 'fas fa-file';
}

async function submitCreateForum() {
    const title      = document.getElementById('createTitle').value.trim();
    const content    = document.getElementById('createContent').value.trim();
    const subject_id = document.getElementById('createSubjectId').value;
    const topic_id   = document.getElementById('createTopicId').value || null;
    const is_pinned  = document.getElementById('createPinned').checked ? 1 : 0;

    if (!title) { toast('Title is required.', 'error'); return; }

    try {
        const res  = await fetch('/teacher/api/forum', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ title, forum_content: content, subject_id, topic_id, is_pinned })
        });
        const data = await res.json();
        if (!res.ok) throw new Error(data.error || 'Failed');

        // Inject new forum into local array and re-render
        data.forum.reply_count   = 0;
        data.forum.done_count    = 0;
        data.forum.enrolled_count = 0;
        data.forum.teacher_pic   = TEACHER_PIC;
        data.forum.teacher_name  = TEACHER_NAME;
        data.forum.last_reply_at = null;
        ALL_FORUMS.unshift(data.forum);

        updateStats();
        applyFilters();
        closeModal('modalCreate');
        toast('Forum created!', 'success');
    } catch(e) { toast(e.message, 'error'); }
}

async function submitEditForum() {
    const title    = document.getElementById('editTitle').value.trim();
    const content  = document.getElementById('editContent').value.trim();
    const topic_id = document.getElementById('editTopicId').value || null;
    if (!title) { toast('Title is required.', 'error'); return; }

    try {
        const res  = await fetch(`/teacher/api/forum/${currentForum.forum_id}`, {
            method: 'PUT',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ title, forum_content: content, topic_id })
        });
        const data = await res.json();
        if (!res.ok) throw new Error(data.error || 'Failed');

        // Update local
        const idx = ALL_FORUMS.findIndex(f => f.forum_id === currentForum.forum_id);
        if (idx > -1) { ALL_FORUMS[idx].title = title; ALL_FORUMS[idx].forum_content = content; ALL_FORUMS[idx].topic_id = topic_id ? Number(topic_id) : null; }
        currentForum.title        = title;
        currentForum.forum_content = content;
        currentForum.topic_id     = topic_id ? Number(topic_id) : null;

        // Re-render thread header elements
        document.getElementById('threadTitle').textContent = title;
        document.getElementById('threadContent').textContent = content;
        const topic = ALL_TOPICS.find(t => t.topic_id === currentForum.topic_id);
        document.getElementById('threadTopicBadge').textContent   = topic ? `Q${topic.quarter_number} · ${topic.topic_name}` : '';
        document.getElementById('threadTopicBadge').style.display = topic ? '' : 'none';

        closeModal('modalEdit');
        toast('Forum updated!', 'success');
    } catch(e) { toast(e.message, 'error'); }
}

async function togglePinThread() {
    if (!currentForum) return;
    const newPin = currentForum.is_pinned ? 0 : 1;
    try {
        const res  = await fetch(`/teacher/api/forum/${currentForum.forum_id}/pin`, {
            method: 'PATCH',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ is_pinned: newPin })
        });
        const data = await res.json();
        if (!res.ok) throw new Error(data.error || 'Failed');

        currentForum.is_pinned = newPin;
        const idx = ALL_FORUMS.findIndex(f => f.forum_id === currentForum.forum_id);
        if (idx > -1) ALL_FORUMS[idx].is_pinned = newPin;

        document.getElementById('threadPinBadge').classList.toggle('hidden', !newPin);
        document.getElementById('btnPinLabel').textContent = newPin ? 'Unpin' : 'Pin';
        updateStats();
        toast(newPin ? 'Forum pinned!' : 'Forum unpinned.', 'success');
    } catch(e) { toast(e.message, 'error'); }
}

async function submitReply() {
    const content  = document.getElementById('replyContent').value.trim();
    const link     = document.getElementById('replyLink').value.trim();
    const fileInput = document.getElementById('replyFile');
    if (!content) { toast('Reply content is required.', 'error'); return; }
    if (!currentForum) return;

    const formData = new FormData();
    formData.append('forum_id',        currentForum.forum_id);
    formData.append('reply_content',   content);
    formData.append('external_link',   link);
    if (replyParentId) formData.append('parent_reply_id', replyParentId);
    if (fileInput.files[0]) formData.append('reply_file', fileInput.files[0]);

    try {
        const res  = await fetch('/teacher/api/forum/reply', { method: 'POST', body: formData });
        const data = await res.json();
        if (!res.ok) throw new Error(data.error || 'Failed');

        closeModal('modalReply');
        toast('Reply posted!', 'success');
        await refreshThread();
    } catch(e) { toast(e.message, 'error'); }
}

async function refreshThread() {
    if (!currentForum) return;
    try {
        const res  = await fetch(`/teacher/api/forum/${currentForum.forum_id}`);
        const data = await res.json();
        if (!res.ok) throw new Error(data.error || 'Failed');

        // Update count in list cache
        const idx = ALL_FORUMS.findIndex(f => f.forum_id === currentForum.forum_id);
        if (idx > -1) {
            ALL_FORUMS[idx].reply_count = data.forum.reply_count || 0;
            ALL_FORUMS[idx].done_count  = data.forum.done_count  || 0;
        }

        currentForum = data.forum;
        renderThread(data);
    } catch(e) { toast(e.message, 'error'); }
}

/* ════════════════════════════════════════════
   UTILS
════════════════════════════════════════════ */
function escHtml(str) {
    if (!str) return '';
    return String(str).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;').replace(/'/g,'&#39;');
}

function formatDate(dt) {
    if (!dt) return '—';
    return new Date(dt).toLocaleDateString('en-PH', { year:'numeric', month:'short', day:'numeric' });
}

function timeAgo(dt) {
    if (!dt) return '—';
    const diff = Math.floor((Date.now() - new Date(dt)) / 1000);
    if (diff < 60)   return 'just now';
    if (diff < 3600) return Math.floor(diff/60)    + 'm ago';
    if (diff < 86400)return Math.floor(diff/3600)  + 'h ago';
    if (diff < 604800)return Math.floor(diff/86400)+ 'd ago';
    return formatDate(dt);
}
